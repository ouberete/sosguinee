        // Animation des chiffres d'impact (Count-up)
        const countUpElements = document.querySelectorAll('.count-up');
        const options = {
          threshold: 0.5 // Déclenche quand 50% de l'élément est visible
        };

        const observer = new IntersectionObserver((entries, observer) => {
          entries.forEach(entry => {
            if (entry.isIntersecting) {
              const target = parseInt(entry.target.dataset.target);
              let current = 0;
              const increment = target / 100; // Ajustez la vitesse de l'animation ici

              const timer = setInterval(() => {
                current += increment;
                if (current >= target) {
                  current = target;
                  clearInterval(timer);
                }
                entry.target.textContent = Math.floor(current);
              }, 20); // Vitesse de l'intervalle

              observer.unobserve(entry.target); // Arrête d'observer une fois l'animation lancée
            }
          });
        }, options);

        countUpElements.forEach(el => {
          observer.observe(el);
        });
      });
    </script>
